```protobuf
service ResponseService {
  rpc ProcessResponse(ProcessResponseRequest) returns (ProcessResponseResponse);
```

### 2. gRPC Server

**Реализованные методы:**
- `ProcessResponse` - обработка ответа
- `GetResponse` - получение ответа по ID
- `ReclassifyResponse` - переклассификация
- `ListResponses` - список ответов с пагинацией
- `GetFormStats` - статистика формы
- `StreamResponses` - server-side streaming

**Код сервера:**

```python
import grpc
from concurrent import futures
import time
import logging
from datetime import datetime
from typing import Dict

from generated import response_service_pb2
from generated import response_service_pb2_grpc

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger(__name__)


class ResponseServiceServicer(response_service_pb2_grpc.ResponseServiceServicer):
    
    def __init__(self):
        self._responses: Dict[str, dict] = {}
    
    def ProcessResponse(self, request, context):
        filling_time = request.submitted_at - request.started_at
        is_suspect = filling_time < 30
        quality_score = 0.15 if is_suspect else 0.95
        
        self._responses[request.response_id] = {
            "response_id": request.response_id,
            "form_id": request.form_id,
            "answers": dict(request.answers),
            "submitted_at": request.submitted_at,
            "quality_score": quality_score,
            "is_suspect": is_suspect,
        }
        
        return response_service_pb2.ProcessResponseResponse(
            is_suspect=is_suspect,
            quality_score=quality_score,
            suspect_reason="filling_time_too_short" if is_suspect else ""
        )
    
    def GetResponse(self, request, context):
        data = self._responses.get(request.response_id)
        if not data:
            context.set_code(grpc.StatusCode.NOT_FOUND)
            return response_service_pb2.ResponseDto()
        
        return response_service_pb2.ResponseDto(
            response_id=data["response_id"],
            form_id=data["form_id"],
            answers=data["answers"],
            submitted_at=data["submitted_at"],
            quality_score=data["quality_score"],
            is_suspect=data["is_suspect"],
        )
    
    def ReclassifyResponse(self, request, context):
        data = self._responses.get(request.response_id)
        if not data:
            context.set_code(grpc.StatusCode.NOT_FOUND)
            return response_service_pb2.ReclassifyResponseResponse()
        
        data["quality_score"] = request.quality_score
        data["is_suspect"] = request.is_suspect
        data["suspect_reason"] = request.suspect_reason
        
        return response_service_pb2.ReclassifyResponseResponse(
            response_id=request.response_id,
            is_suspect=request.is_suspect,
            quality_score=request.quality_score
        )
    
    def ListResponses(self, request, context):
        responses = []
        for data in self._responses.values():
            if data["form_id"] != request.form_id:
                continue
            if request.only_quality and data["is_suspect"]:
                continue
            responses.append(response_service_pb2.ResponseDto(
                response_id=data["response_id"],
                form_id=data["form_id"],
                answers=data["answers"],
                submitted_at=data["submitted_at"],
                quality_score=data["quality_score"],
                is_suspect=data["is_suspect"],
            ))
        
        total = len(responses)
        start = request.offset
        end = start + request.limit
        paginated = responses[start:end]
        
        return response_service_pb2.ListResponsesResponse(
            responses=paginated,
            total=total
        )
    
    def GetFormStats(self, request, context):
        total = 0
        quality = 0
        suspect = 0
        quality_sum = 0.0
        
        for data in self._responses.values():
            if data["form_id"] == request.form_id:
                total += 1
                if data["is_suspect"]:
                    suspect += 1
                else:
                    quality += 1
                    quality_sum += data["quality_score"]
        
        avg_quality = quality_sum / quality if quality > 0 else 0.0
        
        return response_service_pb2.FormStatsDto(
            form_id=request.form_id,
            total_responses=total,
            quality_responses=quality,
            suspect_responses=suspect,
            average_quality=avg_quality,
            average_rating=0.0
        )
    
    def StreamResponses(self, request, context):
        interval = request.interval_sec if request.interval_sec > 0 else 2
        while context.is_active():
            for data in self._responses.values():
                if data["form_id"] == request.form_id:
                    yield response_service_pb2.StreamResponsesResponse(
                        response_id=data["response_id"],
                        form_id=data["form_id"],
                        is_suspect=data["is_suspect"],
                        quality_score=data["quality_score"],
                        submitted_at=data["submitted_at"]
                    )
            time.sleep(interval)


def serve():
    server = grpc.server(futures.ThreadPoolExecutor(max_workers=10))
    response_service_pb2_grpc.add_ResponseServiceServicer_to_server(
        ResponseServiceServicer(), server
    )
    server.add_insecure_port('[::]:50051')
    server.start()
    logger.info("gRPC Server started on port 50051")
    server.wait_for_termination()


if __name__ == "__main__":
    serve()
```

### 3. gRPC Client

import grpc
from datetime import datetime

from generated import response_service_pb2
from generated import response_service_pb2_grpc


class ResponseServiceClient:
    def __init__(self, host='localhost', port=50051):
        self.channel = grpc.insecure_channel(f'{host}:{port}')
        self.stub = response_service_pb2_grpc.ResponseServiceStub(self.channel)
    
    def process_response(self, response_id, form_id, answers):
        request = response_service_pb2.ProcessResponseRequest(
            response_id=response_id,
            form_id=form_id,
            answers=answers,
            started_at=int(datetime.now().timestamp()) - 150,
            submitted_at=int(datetime.now().timestamp())
        )
        return self.stub.ProcessResponse(request)
    
    def get_response(self, response_id):
        request = response_service_pb2.GetResponseRequest(response_id=response_id)
        return self.stub.GetResponse(request)
    
    def reclassify_response(self, response_id, quality_score, is_suspect, suspect_reason):
        request = response_service_pb2.ReclassifyResponseRequest(
            response_id=response_id,
            quality_score=quality_score,
            is_suspect=is_suspect,
            suspect_reason=suspect_reason
        )
        return self.stub.ReclassifyResponse(request)
    
    def list_responses(self, form_id, limit=100, offset=0, only_quality=False):
        request = response_service_pb2.ListResponsesRequest(
            form_id=form_id,
            limit=limit,
            offset=offset,
            only_quality=only_quality
        )
        return self.stub.ListResponses(request)
    
    def get_form_stats(self, form_id):
        request = response_service_pb2.GetFormStatsRequest(form_id=form_id)
        return self.stub.GetFormStats(request)
    
    def stream_responses(self, form_id, interval_sec=2):
        request = response_service_pb2.StreamResponsesRequest(
            form_id=form_id,
            interval_sec=interval_sec
        )
        return self.stub.StreamResponses(request)


def main():
    client = ResponseServiceClient()
    
    print("=== ProcessResponse ===")
    response = client.process_response(
        response_id="RESP-001",
        form_id="FORM-42",
        answers={"name": "Иван", "rating": "5"}
    )
    print(f"Is suspect: {response.is_suspect}")
    print(f"Quality score: {response.quality_score}")
    
    print("\n=== GetResponse ===")
    data = client.get_response("RESP-001")
    print(f"Response ID: {data.response_id}")
    print(f"Is suspect: {data.is_suspect}")
    
    print("\n=== ReclassifyResponse ===")
    result = client.reclassify_response("RESP-001", 0.15, True, "manual_review")
    print(f"Reclassified: is_suspect={result.is_suspect}")
    
    print("\n=== ListResponses ===")
    list_resp = client.list_responses("FORM-42")
    print(f"Total responses: {list_resp.total}")
    
    print("\n=== GetFormStats ===")
    stats = client.get_form_stats("FORM-42")
    print(f"Total: {stats.total_responses}, Quality: {stats.quality_responses}")
    
    print("\n=== StreamResponses ===")
    print("Streaming (press Ctrl+C to stop):")
    try:
        for update in client.stream_responses("FORM-42", interval_sec=2):
            print(f"  -> {update.response_id} | is_suspect={update.is_suspect}")
    except KeyboardInterrupt:
        print("\nStream stopped")


if __name__ == "__main__":
    main()

## Контрольные вопросы

**1. В чём преимущество gRPC над REST?**

- **Производительность:** Protocol Buffers бинарный и компактнее JSON, что даёт меньший размер сообщений и более быструю сериализацию/десериализацию
- **Строгая типизация:** .proto файлы обеспечивают контракт между сервисами и генерацию кода на разных языках
- **Поддержка streaming:** gRPC поддерживает client-side, server-side и bidirectional streaming, что невозможно в REST
- **HTTP/2:** мультиплексирование запросов, сжатие заголовков, keep-alive соединения
- **Автоматическая генерация клиентов:** не нужно писать HTTP-клиенты вручную

**2. Почему Protocol Buffers быстрее JSON?**

- **Бинарный формат:** Protobuf передаёт данные в бинарном виде, а JSON в текстовом. Бинарные данные занимают меньше места и быстрее парсятся
- **Схема на уровне протокола:** Благодаря .proto файлам парсер знает структуру заранее, не нужно анализировать ключи строк
- **Отсутствие накладных расходов:** В Protobuf нет повторяющихся имён полей (вместо этого используются числовые теги)
- **Пример:** JSON `{"name":"Иван","age":25}` (~30 байт) → Protobuf `\x0a\x08Иван\x10\x19` (~12 байт)

**3. Зачем нужен streaming в gRPC?**

- **Real-time обновления:** Получение данных в реальном времени без постоянных polling запросов (например, ленты событий, чаты)
- **Большие объёмы данных:** Передача больших данных частями без загрузки всего в память
- **Двунаправленная коммуникация:** Клиент и сервер могут обмениваться сообщениями одновременно
- **Медиа-стриминг:** Передача аудио/видео потоков
- **Мониторинг и логи:** Непрерывный поток логов или метрик от сервера клиенту
